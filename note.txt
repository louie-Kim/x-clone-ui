... : 객체 전개(Object Spread) , 

{...(tr 
  ? { transformation: [{ width:  `${w}`, height: `${h}` }] }
  : { width: w, height: h }
)}

여기서 삼항 연산자(tr ? A : B)가 먼저 평가된 후, 그 결과 객체(A 또는 B)를 ...이 복사하여 적용합니다.

ImageKit Node.js SDK -> https://github.com/imagekit-developer/imagekit-nodejs?tab=readme-ov-file#file-upload

// Using Callback Function

imagekit.upload({
    file : <url|base_64|binary>, //required
    fileName : "my_file_name.jpg",   //required
    extensions: [
        {
            name: "google-auto-tagging",
            maxTags: 5,
            minConfidence: 95
        }
    ],
    transformation: {
        pre: 'l-text,i-Imagekit,fs-50,l-end',
        post: [
            {
                type: 'transformation',
                value: 'w-100'
            }
        ]
    },
    checks: {`"file.size" < "1mb"`}, // To run server side checks before uploading files. Notice the quotes around file.size and 1mb.
    isPublished: true
}, function(error, result) {
    if(error) console.log(error);
    else console.log(result);
});

// Using Promises

imagekit.upload({
    file : <url|base_64|binary>, //required
    fileName : "my_file_name.jpg",   //required
    extensions: [
        {
            name: "google-auto-tagging",
            maxTags: 5,
            minConfidence: 95
        }
    ],
    transformation: {
        pre: 'l-text,i-Imagekit,fs-50,l-end',
        post: [
            {
                type: 'transformation',
                value: 'w-100'
            }
        ]
    },
    checks={`"file.size" < "1mb"`}
}).then(response => {
    console.log(response);
}).catch(error => {
    console.log(error);
});

--------------------------------------------------------------------------
1. 원본 게시물 A 생성 (Post A)

const postA = await prisma.post.create({
  data: {
    desc: "Original post by User1."
  }
});
 
1. 결과 (데이터베이스에 저장):
 {
  "id": 1,              // Post 모델 id 필드
  "desc": "Original post by User1.",
  "rePostId": null   // 최초생성 게시물 -> null
}

--------------------------------------------------------------------------

2. 리포스트 게시물 B 생성 (Post B) - A를 리포스트함
const postB = await prisma.post.create({
  data: {
    desc: "User2 reposted this post.",
    rePostId: 1  // 원본 게시물 A의 id를 지정해서 저장
  }
});

2. 결과 (데이터베이스에 저장):
{
  "id": 2,          // Post 모델 id 필드
  "desc": "User2 reposted this post.",
  "rePostId": 1   //  원본 게시물의 id 이므로 1부터 시작
}

--------------------------------------------------------------------------


3. 리포스트 게시물 C 생성 (Post C) - A를 리포스트함
const postC = await prisma.post.create({
  data: {
    desc: "User3 also reposted this post.",
    rePostId: 1  // 원본 게시물 A의 id를 지정해서 저장
  }
});

3. 결과 (데이터베이스에 저장):
{
  "id": 3,            // Post 모델 id 필드
  "desc": "User3 also reposted this post.",
  "rePostId": 1
}


리포스트된 게시물들을 조회하기 (Post A의 rePosts)

* findUnique 하나의 고유한 레코드(PK)를 조회하는 함수 

const originalPost = await prisma.post.findUnique({
  where: { id: 1 },           // id가 1인 게시물을 조회
  include: { rePosts: true }  // 리포스트된 게시물들을 포함해서 가져오기
});

console.log(originalPost);

{

      "rePostId": 1
    },

예제 상황 (댓글 기능 작동 과정 보기)

1. 원본 게시물 A 생성 (Post A)

const postA = await prisma.post.create({
  data: {
    desc: "Original post by User1."
  }
});

{
  "id": 1,
  "desc": "Original post by User1.",
  "parentPostId": null  
}

2. 댓글 작성하기 (Post B - A에 대한 댓글)

const postB = await prisma.post.create({
  data: {
    desc: "User2 commented on Post A.",
    parentPostId: 1  // 댓글이 달린 원본 게시물의 id (Post A)
  }
});

{
  "id": 2,
  "desc": "User2 commented on Post A.",
  "parentPostId": 1  // Post A의 id
}

3. 또 다른 댓글 작성하기 (Post C - A에 대한 댓글)
const postC = await prisma.post.create({
  data: {
    desc: "User3 also commented on Post A.",
    parentPostId: 1  // 댓글이 달린 원본 게시물의 id (Post A)
  }
});

{
  "id": 3,
  "desc": "User3 also commented on Post A.",
  "parentPostId": 1  // Post A의 id
}

댓글 목록 조회하기 (원본 게시물 A의 comments)

const originalPost = await prisma.post.findUnique({
  where: { id: 1 },
  include: { comments: true }  // 모든 댓글을 포함해서 가져오기
});

console.log(originalPost);

{
  "id": 1,
  "desc": "Original post by User1.",
  // 처음 생성된 게시물(=원본 게시물) 에는 댓글이 없기 때문에 parentPostId 필드는 null로 설정
  "parentPostId": null,
  "comments": [
    {
      "id": 2,
      "desc": "User2 commented on Post A.",
      "parentPostId": 1
    },
    {
      "id": 3,
      "desc": "User3 also commented on Post A.",
      "parentPostId": 1
    }
  ]
}
